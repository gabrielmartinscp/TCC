CREATE DATABASE IF NOT EXISTS sistema_orcamento;
USE sistema_orcamento;

CREATE TABLE clientes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(150) NOT NULL,
    email VARCHAR(150),
    telefone VARCHAR(30),
    deletado BOOLEAN DEFAULT FALSE
);

CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(150) NOT NULL,
    senha VARCHAR(255) NOT NULL,
    tipo VARCHAR(50),
    email VARCHAR(150) UNIQUE
);

CREATE TABLE fornecedor (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(150) NOT NULL,
    email VARCHAR(150),
    telefone VARCHAR(30),
    deletado BOOLEAN DEFAULT FALSE
);

CREATE TABLE insumo (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(150) NOT NULL,
    unidade VARCHAR(50),
    preco_atual DECIMAL(10,2),
    id_fornecedor INT,

    CONSTRAINT fk_insumo_fornecedor
        FOREIGN KEY (id_fornecedor)
        REFERENCES fornecedor(id)
        ON DELETE SET NULL
);

CREATE TABLE projeto (
    id_projeto INT AUTO_INCREMENT PRIMARY KEY,
    id_cliente INT NOT NULL,

    nome VARCHAR(150) NOT NULL,
    tipo_obra VARCHAR(100),
    descricao TEXT,
    data_inicio DATE,
    data_fim DATE,

    CONSTRAINT fk_projeto_cliente
        FOREIGN KEY (id_cliente)
        REFERENCES clientes(id)
        ON DELETE CASCADE
);

CREATE TABLE orcamento (
    id_orcamento INT AUTO_INCREMENT PRIMARY KEY,

    id_projeto INT NOT NULL,
    id_usuario INT NOT NULL,

    data_criacao DATE,
    custo_total DECIMAL(12,2),
    valor_final DECIMAL(12,2),
    margem_lucro DECIMAL(5,2),
    impostos DECIMAL(5,2),

    CONSTRAINT fk_orcamento_projeto
        FOREIGN KEY (id_projeto)
        REFERENCES projeto(id_projeto)
        ON DELETE CASCADE,

    CONSTRAINT fk_orcamento_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuarios(id)
        ON DELETE CASCADE
);

CREATE TABLE item_orcamento (
    id_item INT AUTO_INCREMENT PRIMARY KEY,

    id_orcamento INT NOT NULL,
    id_insumo INT NOT NULL,

    quantidade DECIMAL(10,2),
    preco_unitario DECIMAL(10,2),
    custo_total_item DECIMAL(12,2),

    CONSTRAINT fk_item_orcamento
        FOREIGN KEY (id_orcamento)
        REFERENCES orcamento(id_orcamento)
        ON DELETE CASCADE,

    CONSTRAINT fk_item_insumo
        FOREIGN KEY (id_insumo)
        REFERENCES insumo(id_insumo)
        ON DELETE CASCADE
);

CREATE TABLE historico_preco_insumo (
    id_historico INT AUTO_INCREMENT PRIMARY KEY,

    id_insumo INT NOT NULL,
    id_usuario INT NOT NULL,

    preco DECIMAL(10,2),
    data_inicio_vigencia DATE,
    data_fim_vigencia DATE NULL,

    CONSTRAINT fk_historico_insumo
        FOREIGN KEY (id_insumo)
        REFERENCES insumo(id_insumo)
        ON DELETE CASCADE,

    CONSTRAINT fk_historico_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuarios(id)
        ON DELETE CASCADE
);

CREATE TABLE execucao_projeto (
    id_execucao INT AUTO_INCREMENT PRIMARY KEY,

    id_projeto INT NOT NULL,

    custo_real DECIMAL(12,2),
    data_registro DATE,

    CONSTRAINT fk_execucao_projeto
        FOREIGN KEY (id_projeto)
        REFERENCES projeto(id_projeto)
        ON DELETE CASCADE
);

CREATE TABLE alerta (
    id_alerta INT AUTO_INCREMENT PRIMARY KEY,

    tipo_alerta VARCHAR(100),
    descricao TEXT,
    data DATE,

    id_insumo INT NULL,
    id_projeto INT NULL,

    CONSTRAINT fk_alerta_insumo
        FOREIGN KEY (id_insumo)
        REFERENCES insumo(id_insumo)
        ON DELETE SET NULL,

    CONSTRAINT fk_alerta_projeto
        FOREIGN KEY (id_projeto)
        REFERENCES projeto(id_projeto)
        ON DELETE SET NULL
);
